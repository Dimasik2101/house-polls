      <h3 id="form-title" style="margin-bottom: 14px;">Новый опрос</h3>

      <input type="hidden" id="edit-id">

      <input type="text" id="poll-question" placeholder="Суть вопроса (напр. Замена ворот во дворе)">

      

      <div id="options-container">

  <script>

    const DB_URL = "https://house-polls-default-rtdb.europe-west1.firebasedatabase.app";


    const tg = window.Telegram.WebApp;

    tg.expand();

    const user = tg.initDataUnsafe?.user || { id: Math.floor(Math.random()*100000), first_name: "Тест-Жилец" };


    let polls = [];


    async function loadPolls() {

      try {

        const res = await fetch(`${DB_URL}/polls.json`);

        const data = await res.json();

        polls = data ? Object.values(data).sort((a, b) => b.id - a.id) : [];

      } catch (e) {

        console.error("Ошибка загрузки:", e);

      }

      document.getElementById('loader').classList.add('hidden');

      render();

    }


    async function syncPollToDB(poll) {

      await fetch(`${DB_URL}/polls/${poll.id}.json`, {

        method: 'PUT',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(poll)

      });

      await loadPolls();

    }


    function switchTab(tab) {

      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));

      document.getElementById('tab-btn-' + tab).classList.add('active');

      

      ['active', 'create', 'history'].forEach(v => {

        document.getElementById('view-' + v).classList.toggle('hidden', v !== tab);

      });

      if (tab === 'create' && !document.getElementById('edit-id').value) resetForm();

      render();

    }


    function addOptionInput(val = '') {

      const inp = document.createElement('input');

      inp.type = 'text';

      inp.className = 'poll-opt';

      inp.placeholder = 'Следующий вариант';

      inp.value = val;

      document.getElementById('options-container').appendChild(inp);

    }


    function resetForm() {

      document.getElementById('form-title').innerText = 'Новый опрос';

      document.getElementById('edit-id').value = '';

      document.getElementById('poll-question').value = '';

      document.getElementById('poll-anon').checked = false;

      document.getElementById('options-container').innerHTML = `

        <input type="text" class="poll-opt" placeholder="Вариант 1">

        <input type="text" class="poll-opt" placeholder="Вариант 2">

      `;

    }


    async function savePoll() {

      const id = document.getElementById('edit-id').value;

      const question = document.getElementById('poll-question').value.trim();

      const isAnon = document.getElementById('poll-anon').checked;

      const opts = Array.from(document.querySelectorAll('.poll-opt'))

        .map(i => i.value.trim())

        .filter(Boolean);


      if (!question || opts.length < 2) return alert('Введите вопрос и минимум 2 варианта ответа!');


      let targetPoll;

      if (id) {

        targetPoll = polls.find(p => p.id == id);

        targetPoll.question = question;

        targetPoll.isAnon = isAnon;

        targetPoll.options = opts.map((text, idx) => ({

          text,

          voters: (targetPoll.options[idx] && targetPoll.options[idx].voters) ? targetPoll.options[idx].voters : []

        }));

      } else {

        targetPoll = {

          id: Date.now(),

          question,

          isAnon,

          status: 'active',

          date: new Date().toLocaleDateString('ru-RU'),

          options: opts.map(text => ({ text, voters: [] }))

        };

      }


      document.getElementById('loader').classList.remove('hidden');

      await syncPollToDB(targetPoll);

      resetForm();

      switchTab('active');

    }


    async function vote(pollId, optIdx) {

      const poll = polls.find(p => p.id === pollId);

      if (!poll || poll.status !== 'active') return;

      

      poll.options.forEach(o => {

        if (!o.voters) o.voters = [];

        o.voters = o.voters.filter(v => v.id !== user.id);

      });

      poll.options[optIdx].voters.push({ id: user.id, name: user.first_name });

      

      render();

      await syncPollToDB(poll);

    }


    function editPoll(id) {

      const poll = polls.find(p => p.id === id);

      switchTab('create');

      document.getElementById('form-title').innerText = 'Редактирование';

      document.getElementById('edit-id').value = poll.id;

      document.getElementById('poll-question').value = poll.question;

      document.getElementById('poll-anon').checked = poll.isAnon;

      const cont = document.getElementById('options-container');

      cont.innerHTML = '';

      poll.options.forEach(o => addOptionInput(o.text));

    }


    async function archivePoll(id) {

      const poll = polls.find(p => p.id === id);

      poll.status = 'history';

      await syncPollToDB(poll);

    }


    function render() {

      const activeEl = document.getElementById('view-active');

      const historyEl = document.getElementById('view-history');

      activeEl.innerHTML = '';

      historyEl.innerHTML = '';


      const activePolls = polls.filter(p => p.status === 'active');

      const historyPolls = polls.filter(p => p.status !== 'active');


      if (activePolls.length === 0) {

        activeEl.innerHTML = `<div class="card" style="text-align: center; color: var(--hint);">Пока нет активных опросов. Нажмите «+ Опрос», чтобы создать первый!</div>`;

      }

      if (historyPolls.length === 0) {

        historyEl.innerHTML = `<div class="card" style="text-align: center; color: var(--hint);">История опросов пока пуста.</div>`;

      }


      polls.forEach(poll => {

        const totalVotes = poll.options.reduce((sum, o) => sum + (o.voters ? o.voters.length : 0), 0);

        const card = document.createElement('div');

        card.className = 'card';


        let optionsHtml = poll.options.map((opt, idx) => {

          const voters = opt.voters || [];

          const pct = totalVotes ? Math.round((voters.length / totalVotes) * 100) : 0;

          const isVoted = voters.some(v => v.id === user.id);

          const names = !poll.isAnon && voters.length 

            ? `<div class="voter-names">Проголосовали: ${voters.map(v => v.name).join(', ')}</div>` 

            : '';

          return `

            <div class="option-row ${isVoted ? 'voted' : ''}" onclick="vote(${poll.id}, ${idx})">

              <div>

                <div style="font-weight: 500;">${opt.text}</div>

                ${names}

              </div>

              <strong style="color: var(--accent);">${voters.length} (${pct}%)</strong>

            </div>

          `;

        }).join('');


        const actionsHtml = poll.status === 'active' ? `

          <div class="actions">

            <button class="btn btn-sec" style="margin-top:0;" onclick="editPoll(${poll.id})">✏️ Изменить</button>

            <button class="btn btn-sec" style="margin-top:0;" onclick="archivePoll(${poll.id})">📦 В архив</button>

          </div>

        ` : `<div style="font-size: 12px; color: var(--hint); margin-top: 10px;">Архив • ${poll.date}</div>`;


        card.innerHTML = `

          <span class="badge ${poll.isAnon ? 'anon' : ''}">

            ${poll.isAnon ? '🕵️ Анонимный опрос' : '👁 Открытый опрос'}

          </span>

          <h4>${poll.question}</h4>

          ${optionsHtml}

          ${actionsHtml}

        `;


        if (poll.status === 'active') activeEl.appendChild(card);

        else historyEl.appendChild(card);

      });

    }


    loadPolls();

    setInterval(loadPolls, 5000);

  </script>
